module uart_tb;
reg clk, rst;
reg [7:0]data_in;
reg wr_en;
reg rdy_clr;
wire rdy;
wire busy;
wire [7:0] data_out;
uart_top dut(
 clk, rst, wr_en,rdy_clr, data_in, busy,data_out,rdy
);
always #5 clk=~clk;
initial begin
    {clk,wr_en,data_in,rdy_clr}=0;
    rst=1;
    #20 rst=0;
end

task send(input [7:0]din);
    begin
    @(posedge clk);
        data_in=din;
        wr_en=1'b1;
    @(posedge clk)
        wr_en=0;
    end 
endtask

task clear_rdy;
    begin
        @(posedge clk)
        rdy_clr=1'b1;
        @(posedge clk)
        rdy_clr=1'b0;
    end
  endtask
always @(posedge rdy)
    $display("t=%0t byte received: %h", $time, data_out);
    
initial begin
    wait(!rst);
    send(8'h00);//tesing all zeros input
    wait(busy);
    wait(!busy);
    wait (rdy);
    clear_rdy;
    send(8'hff);// testing all ones input
    wait(busy);
    wait(!busy);
    wait(rdy);
    clear_rdy;
    send(8'haa);//testing alternating input
    wait(busy);
    wait(!busy);
    wait(rdy);
    clear_rdy;
    send(8'h18);//random input
    wait(busy);
    wait(!busy);
    wait(rdy);
    clear_rdy;
    //testing back to back inputs
     send(8'hab);
     wait(busy);
     wait(!busy);
     send(8'hcd);
     wait (busy);
     wait(!busy);
     send(8'hef);
     wait(rdy);
     clear_rdy;
     wait(rdy);
     clear_rdy;
     wait(rdy);
     clear_rdy;
    $finish;
end
  
endmodule
